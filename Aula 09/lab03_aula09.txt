# Objetivo: (1) comparar os dois formatos de função de pertinência,
#           (2) testar o controlador com várias entradas,
#           (3) ver a "superfície" que mostra todas as respostas possíveis.

!pip install -q scikit-fuzzy

import os
import numpy as np
import skfuzzy as fuzz
from skfuzzy import control as ctrl
import matplotlib.pyplot as plt
from mpl_toolkits.mplot3d import Axes3D  
os.makedirs("imagens", exist_ok=True)

x = np.arange(0, 10.01, 0.01)


tri_ruim  = fuzz.trimf(x, [0, 0, 5])
tri_aceit = fuzz.trimf(x, [0, 5, 10])
tri_exc   = fuzz.trimf(x, [5, 10, 10])

trap_ruim  = fuzz.trapmf(x, [0, 0, 2, 4])
trap_aceit = fuzz.trapmf(x, [2, 4, 6, 8])
trap_exc   = fuzz.trapmf(x, [6, 8, 10, 10])

fig, (a1, a2) = plt.subplots(2, 1, figsize=(8, 6), sharex=True)
for y, r in ((tri_ruim, "Ruim"), (tri_aceit, "Aceitável"), (tri_exc, "Excelente")):
    a1.plot(x, y, label=r)
a1.set_title("Serviço - funções triangulares")
a1.legend()
for y, r in ((trap_ruim, "Ruim"), (trap_aceit, "Aceitável"), (trap_exc, "Excelente")):
    a2.plot(x, y, label=r)
a2.set_title("Serviço - funções trapezoidais")
a2.legend()
for a in (a1, a2):
    a.set_ylabel("Pertinência")
a2.set_xlabel("Nota do serviço (0 a 10)")
plt.tight_layout()
plt.savefig("imagens/lab03_tri_vs_trap.png", dpi=150)
plt.show()

qualidade = ctrl.Antecedent(np.arange(0, 11, 1), "qualidade")
servico   = ctrl.Antecedent(np.arange(0, 11, 1), "servico")
gorjeta   = ctrl.Consequent(np.arange(0, 26, 1), "gorjeta")

qualidade.automf(3, names=["ruim", "razoavel", "otima"])
servico.automf(3,   names=["ruim", "aceitavel", "excelente"])
gorjeta["baixa"] = fuzz.trimf(gorjeta.universe, [0, 0, 13])
gorjeta["media"] = fuzz.trimf(gorjeta.universe, [0, 13, 25])
gorjeta["alta"]  = fuzz.trimf(gorjeta.universe, [13, 25, 25])

regras = [
    ctrl.Rule(servico["excelente"] | qualidade["otima"], gorjeta["alta"]),
    ctrl.Rule(servico["aceitavel"],                      gorjeta["media"]),
    ctrl.Rule(servico["ruim"] & qualidade["ruim"],       gorjeta["baixa"]),
]
sistema = ctrl.ControlSystem(regras)
sim = ctrl.ControlSystemSimulation(sistema)

# Cada par é (qualidade, serviço). Escolhi casos nos extremos e no meio
# para ver se a saída faz sentido: piores notas -> gorjeta baixa,
# melhores notas -> gorjeta alta.

casos = [(0, 0), (2, 3), (5, 5), (6.5, 9.8), (10, 10)]
print("qualidade | servico | gorjeta (%)")
for q, s in casos:
    sim.input["qualidade"] = q
    sim.input["servico"]   = s
    sim.compute()                       
    print(f"{q:9} | {s:7} | {sim.output['gorjeta']:.2f}")

xs = np.arange(1, 11, 1)
ys = np.arange(1, 11, 1)
X, Y = np.meshgrid(xs, ys)             
Z = np.zeros_like(X, dtype=float)      


for i in range(X.shape[0]):
    for j in range(X.shape[1]):
        sim.input["qualidade"] = X[i, j]
        sim.input["servico"]   = Y[i, j]
        sim.compute()
        Z[i, j] = sim.output["gorjeta"]

fig = plt.figure(figsize=(8, 6))
ax = fig.add_subplot(111, projection="3d")
ax.plot_surface(X, Y, Z, cmap="viridis", linewidth=0.4, antialiased=True)
ax.set_xlabel("Qualidade da comida")
ax.set_ylabel("Serviço")
ax.set_zlabel("Gorjeta (%)")
ax.set_title("Superfície de controle fuzzy")
plt.savefig("imagens/lab03_superficie.png", dpi=150)
plt.show()
